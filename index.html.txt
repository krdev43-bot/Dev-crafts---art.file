    <script>
        let orders = JSON.parse(localStorage.getItem('dev_crafts_orders')) || [];
        let settings = JSON.parse(localStorage.getItem('dev_crafts_settings')) || {
            propImg: '',
            wmImg: '',
            wmOpacity: '0.15'
        };
        let currentReceiptOrder = null;

        window.onload = function() {
            applySettings();
            renderOrders();
        };

        function applySettings() {
            if(settings.propImg) {
                document.getElementById('headerPropImg').src = settings.propImg;
            }
            if(settings.wmImg) {
                document.documentElement.style.setProperty('--wm-image', `url(${settings.wmImg})`);
            }
            document.documentElement.style.setProperty('--wm-opacity', settings.wmOpacity || '0.15');
            document.getElementById('wmOpacityRange').value = settings.wmOpacity || '0.15';
        }

        function updateProfilePic(input) {
            if (input.files && input.files[0]) {
                let reader = new FileReader();
                reader.onload = function(e) {
                    settings.propImg = e.target.result;
                    localStorage.setItem('dev_crafts_settings', JSON.stringify(settings));
                    applySettings();
                };
                reader.readAsDataURL(input.files[0]);
            }
        }

        function updateWatermark(input) {
            if (input.files && input.files[0]) {
                let reader = new FileReader();
                reader.onload = function(e) {
                    settings.wmImg = e.target.result;
                    localStorage.setItem('dev_crafts_settings', JSON.stringify(settings));
                    applySettings();
                };
                reader.readAsDataURL(input.files[0]);
            }
        }

        function changeWmOpacity(val) {
            settings.wmOpacity = val;
            localStorage.setItem('dev_crafts_settings', JSON.stringify(settings));
            applySettings();
        }

        function getNextOrderId() {
            let num = orders.length + 1;
            return 'DEV-' + String(num).padStart(3, '0');
        }

        function clearForm() {
            document.getElementById('editIndex').value = '';
            document.getElementById('modalTitle').innerText = 'नया ऑर्डर जोड़ें';
            document.getElementById('orderId').value = getNextOrderId();
            document.getElementById('custName').value = '';
            document.getElementById('custPhone').value = '';
            document.getElementById('orderDate').value = new Date().toISOString().split('T')[0];
            document.getElementById('deliveryDate').value = '';
            document.getElementById('workDetails').value = '';
            document.getElementById('totalAmt').value = 0;
            document.getElementById('advAmt').value = 0;
            document.getElementById('pendingAmt').value = 0;
            document.getElementById('orderStatus').value = 'Pending';
            document.getElementById('modalImgPreview').classList.add('d-none');
            document.getElementById('modalImgPreview').src = '';
            document.getElementById('designImageInput').value = '';
        }

        function calcPending() {
            let total = parseFloat(document.getElementById('totalAmt').value) || 0;
            let adv = parseFloat(document.getElementById('advAmt').value) || 0;
            document.getElementById('pendingAmt').value = total - adv;
        }

        function previewImage(input, targetId) {
            if (input.files && input.files[0]) {
                let reader = new FileReader();
                reader.onload = function(e) {
                    let img = document.getElementById(targetId);
                    img.src = e.target.result;
                    img.classList.remove('d-none');
                };
                reader.readAsDataURL(input.files[0]);
            }
        }

        function saveOrder() {
            let name = document.getElementById('custName').value.trim();
            let phone = document.getElementById('custPhone').value.trim();
            if(!name || !phone) {
                alert('कृपया कस्टमर का नाम और नंबर भरें!');
                return;
            }

            let index = document.getElementById('editIndex').value;
            let imgData = document.getElementById('modalImgPreview').src;

            let orderData = {
                id: document.getElementById('orderId').value,
                name: name,
                phone: phone,
                orderDate: document.getElementById('orderDate').value,
                deliveryDate: document.getElementById('deliveryDate').value,
                work: document.getElementById('workDetails').value,
                total: parseFloat(document.getElementById('totalAmt').value) || 0,
                adv: parseFloat(document.getElementById('advAmt').value) || 0,
                pending: parseFloat(document.getElementById('pendingAmt').value) || 0,
                status: document.getElementById('orderStatus').value,
                image: imgData && !imgData.includes('window.location') ? imgData : 'https://via.placeholder.com/300x200?text=No+Design+Photo'
            };

            if(index === '') {
                orders.unshift(orderData);
            } else {
                orders[index] = orderData;
            }

            localStorage.setItem('dev_crafts_orders', JSON.stringify(orders));
            bootstrap.Modal.getInstance(document.getElementById('orderModal')).hide();
            renderOrders();
        }

        function renderOrders() {
            let list = document.getElementById('ordersList');
            let search = document.getElementById('searchInput').value.toLowerCase();
            list.innerHTML = '';

            let filtered = orders.filter(o => 
                o.name.toLowerCase().includes(search) || 
                o.phone.includes(search) || 
                o.id.toLowerCase().includes(search)
            );

            if(filtered.length === 0) {
                list.innerHTML = '<div class="col-12 text-center text-muted py-5">कोई ऑर्डर नहीं मिला।</div>';
                return;
            }

            filtered.forEach((o, i) => {
                let statusClass = o.status === 'Completed' ? 'bg-success' : o.status === 'In Progress' ? 'bg-primary' : o.status === 'Delivered' ? 'bg-dark' : 'bg-warning text-dark';
                
                let card = `
                <div class="col-md-6 col-lg-4">
                    <div class="card watermarked-area">
                        <div class="card-body watermarked-content">
                            <div class="d-flex justify-content-between align-items-start mb-2">
                                <div>
                                    <span class="badge bg-secondary mb-1">${o.id}</span>
                                    <h5 class="card-title m-0 fw-bold">${o.name}</h5>
                                    <small class="text-muted"><i class="bi bi-telephone"></i> ${o.phone}</small>
                                </div>
                                <span class="badge ${statusClass} status-badge">${o.status}</span>
                            </div>
                            
                            <p class="card-text small text-truncate mb-2"><strong>काम:</strong> ${o.work || 'N/A'}</p>
                            
                            <div class="row text-center mb-3 g-1">
                                <div class="col-4 border-end"><small>कुल</small><br><strong>₹${o.total}</strong></div>
                                <div class="col-4 border-end text-success"><small>एडवांस</small><br><strong>₹${o.adv}</strong></div>
                                <div class="col-4 text-danger"><small>बकाया</small><br><strong>₹${o.pending}</strong></div>
                            </div>

                            <div class="d-flex gap-1">
                                <button class="btn btn-sm btn-outline-primary flex-fill" onclick="openReceipt(${orders.indexOf(o)})"><i class="bi bi-receipt"></i> बिल देखें</button>
                                <button class="btn btn-sm btn-outline-secondary" onclick="editOrder(${orders.indexOf(o)})"><i class="bi bi-pencil"></i></button>
                                <button class="btn btn-sm btn-outline-danger" onclick="deleteOrder(${orders.indexOf(o)})"><i class="bi bi-trash"></i></button>
                            </div>
                        </div>
                    </div>
                </div>`;
                list.innerHTML += card;
            });
        }

        function editOrder(index) {
            let o = orders[index];
            document.getElementById('editIndex').value = index;
            document.getElementById('modalTitle').innerText = 'ऑर्डर एडिट करें';
            document.getElementById('orderId').value = o.id;
            document.getElementById('custName').value = o.name;
            document.getElementById('custPhone').value = o.phone;
            document.getElementById('orderDate').value = o.orderDate;
            document.getElementById('deliveryDate').value = o.deliveryDate;
            document.getElementById('workDetails').value = o.work;
            document.getElementById('totalAmt').value = o.total;
            document.getElementById('advAmt').value = o.adv;
            document.getElementById('pendingAmt').value = o.pending;
            document.getElementById('orderStatus').value = o.status;
            
            if(o.image) {
                let img = document.getElementById('modalImgPreview');
                img.src = o.image;
                img.classList.remove('d-none');
            }
            new bootstrap.Modal(document.getElementById('orderModal')).show();
        }

        function deleteOrder(index) {
            if(confirm('क्या आप सच में इस ऑर्डर को डिलीट करना चाहते हैं?')) {
                orders.splice(index, 1);
                localStorage.setItem('dev_crafts_orders', JSON.stringify(orders));
                renderOrders();
            }
        }

        function openReceipt(index) {
            let o = orders[index];
            currentReceiptOrder = o;
            document.getElementById('rOrderId').innerText = o.id;
            document.getElementById('rDate').innerText = 'तारीख: ' + (o.orderDate || 'N/A');
            document.getElementById('rCustName').innerText = o.name;
            document.getElementById('rCustPhone').innerText = o.phone;
            document.getElementById('rWorkDetails').innerText = o.work || 'कोई विवरण नहीं';
            document.getElementById('rTotal').innerText = '₹' + o.total;
            document.getElementById('rAdv').innerText = '₹' + o.adv;
            document.getElementById('rPending').innerText = '₹' + o.pending;
            document.getElementById('rImage').src = o.image;

            new bootstrap.Modal(document.getElementById('receiptModal')).show();
        }

        function sendWhatsApp() {
            if(!currentReceiptOrder) return;
            let o = currentReceiptOrder;
            let msg = `नमस्ते ${o.name} जी,\n\n` +
                      `आपका *Dev Crafts & CNC Design* में ऑर्डर दर्ज कर लिया गया है।\n\n` +
                      `*ऑर्डर ID:* ${o.id}\n` +
                      `*विवरण:* ${o.work}\n` +
                      `*कुल रकम:* ₹${o.total}\n` +
                      `*एडवांस:* ₹${o.adv}\n` +
                      `*बकाया राशि:* ₹${o.pending}\n\n` +
                      `धन्यवाद!\n` +
                      `*देव कुमार (Dev Crafts)*\n` +
                      `संपर्क: 9279108212`;
            
            let url = `https://wa.me/91${o.phone}?text=${encodeURIComponent(msg)}`;
